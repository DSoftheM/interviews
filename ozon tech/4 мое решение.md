interface IBookingTourService
{
    // аргументы - UserInfo, FlightInfo, RentInfo, TourInfo
    public bool BookTour(string userId, DateTime start, DateTime end, int hotelId, string roomType, string flightClass,
        int flightNumber, int airCompanyId);
}

public class BookingTourService : IBookingTourService
{
    private readonly IAirWaysService _airWaysService;
    private readonly IHotelService _hotelService;
    private readonly IUserHttpClient _userHttpClient;
    private readonly IUnitOfWork _unitOfWork;
    private readonly IBookingTourRepository _tourRepository;

    // Проверить гайдлайны по статик полям
    // не должны быть static
    // спрятать внутрь метода BookTour
    private static Guid _reservationIdAir;
    private static Guid _reservationIdHotel;

    // Можно удалить, т. к. не используется
    public static long _countBooked;

    // Primary constructor
    public BookingTourService(IAirWaysService airWaysService, IHotelService hotelService,
        IUserHttpClient userHttpClient, IUnitOfWork unitOfWork, IBookingTourRepository tourRepository)
    {
        _airWaysService = airWaysService;
        _hotelService = hotelService;
        _userHttpClient = userHttpClient;
        _unitOfWork = unitOfWork;
        _tourRepository = tourRepository;
    }


    // async Task
    public bool BookTour(string userId, DateTime start, DateTime end, int hotelId, string roomType, string flightClass,
        int airCompanyId, int flightnumber)
    {
        try
        {
            var flightInfo = new FlightInfo(Enum.Parse<FlightClass>(flightClass), airCompanyId, flightnumber); // naming
            var resultBookFlight = _airWaysService.Book(GetUser(userId), flightInfo);

            var rentInfo = new RentInfo(start, end, hotelId, Enum.Parse<RoomType>(roomType));
            var resultBookHotel = _hotelService.Book(GetUser(userId), rentInfo);

            // await
            // Непонятно, кто из них выполнится первым, и не получится правильно обработать CancelBook
            Task.WhenAll(resultBookFlight, resultBookHotel);

            _reservationIdAir = resultBookFlight.Result;
            _reservationIdHotel = resultBookHotel.Result;
        }
        catch (BookFlightException)
        {
            // await
            _hotelService.CancelBook(_reservationIdAir); // _reservationIdHotel
            // throw;
            return false;
        }
        catch (BookHotelException)
        {
            // await
            _airWaysService.CancelBook(_reservationIdHotel); // _reservationIdAir
            // throw;
            return false;
        }
        catch (Exception ex)
        {
            // throw;
            // 
            throw new ApplicationException(ex.Message, ex);
        }
        finally
        {
            // ExecuteAsync
            // Тур создастся в любом случае, независимо от того, забронировались ли сущности
            // await
            _unitOfWork.Execute(async () =>
            {
                // await _tourRepository.Save(tour)
                await _tourRepository.Save(_reservationIdAir, _reservationIdHotel, GetUser(userId));
            }).RunSynchronously(); // Remove RunSynchronously
            _countBooked++; // Interlocked.Increment();
        }

        return true; // удалить
    }

    // Task<UserInfo>
    private UserInfo GetUser(string userId)
    {
        // remove long.Parse
        // await 
        // try-catch
        return _userHttpClient.GetUserById(long.Parse(userId)).Result;
    }
}

public interface IAirWaysService
{
    Task<Guid> Book(UserInfo user, FlightInfo flightInfo, CancelationToken ct = CancelationToken.None);
    Task CancelBook(Guid reservationId, CancelationToken ct = CancelationToken.None);
}

public interface IHotelService
{
    Task<Guid> Book(UserInfo user, RentInfo rentInfo, CancelationToken ct = CancelationToken.None);
    Task CancelBook(Guid reservationId, CancelationToken ct = CancelationToken.None);
}

public interface IBookingTourRepository
{
    Task Save(Guid reservationAirId, Guid reservationHotelId, UserInfo user,
        CancelationToken ct = CancelationToken.None);
}

public interface IUserHttpClient
{
    Task<UserInfo> GetUserById(long userId, CancelationToken ct = CancelationToken.None);
}

public interface IUnitOfWork
{
    Task Execute(Func<Task> action);
}

public class BookFlightException : Exception

}

public class BookHotelException : Exception
{
}

public record FlightInfo(FlightClass flightClass, long airCompanyId, long flightNumber);

public record RentInfo(DateTimeOffset start, DateTimeOffset end, long hotelid, RoomType roomType);

public record UserInfo(long userId, string userName);

public enum FlightClass
