```sql
drop table salary;  
  
-- Создание таблицы  
CREATE TABLE salary  
(  
    id     SERIAL PRIMARY KEY, -- SERIAL вместо int + автоинкремент  
    salary NUMERIC(12, 2) NOT NULL  
);  
  
-- Вставка дополнительных данных  
INSERT INTO salary (id, salary)  
VALUES (5, 400),  
       (6, 500),  
       (7, 300),  
       (8, 600),  
       (9, 200),  
       (10, 700),  
       (11, 500),  
       (12, 800),  
       (13, 100),  
       (14, 900),  
       (15, 900),  
       (16, 1000);
       
SELECT salary  
FROM (SELECT salary,  
             DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk  
      FROM salary) t1  
WHERE rnk = 3;

SELECT DISTINCT salary  
FROM salary  
ORDER BY salary DESC  
OFFSET 3 - 1 LIMIT 1;

select distinct on (s.salary) salary  
from salary s  
order by salary desc  
offset 3 - 1 limit 1;
```
