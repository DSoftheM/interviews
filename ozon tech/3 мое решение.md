using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace ConsoleApp
{
    public sealed class UserEnricher
    {
        private readonly IDbRepository _dbRepository;
        private readonly ICacheRepository _cacheRepository;
        private readonly IEventSaver _eventSaver;
        public UserEnricher(IDbRepository dbRepository, ICacheRepository cacheRepository, IEventSaver eventSaver)
        {
            _cacheRepository = cacheRepository;
            _eventSaver = eventSaver;
            _dbRepository = dbRepository;
        }
        
        public async Task Enrich(IEnumerable<UserEvent> events)
        {
            foreach (var event of events) {
                EnrichUserEvent(event);
            }

            ProccessFromQueue();
        }

        private List<UserEvent> _idsForBatching = new();

        public async Task ProccessFromQueue() {
            foreach (var event of _idsForBatching) {
                // [ [1,2,3..99,100], [1,2,3..99,100]  ] // 640
            }

            for (var i = 0; i < Math.Ceil(_idsForBatching.Length / 100); i++) 
            {
                var events = new Dictionary<long, UserEvent>();
                for (var j = i * 100; j < ((i + 1) * 100) && j < _idsForBatching.Length; j++) {
                    var event = _idsForBatching[j];
                    events.Add(event.Id);                    
                }
                
                Dictionary<long,string> namesFromDb = await _dbRepository.GetFullName(event.Keys);

                foreach (var result in namesFromDb) {
                    events[result.Key].SetFullName(result.Value);
                    _eventSaver.Save(events[result.Key]);
                }
            }
        }

        public async Task EnrichUserEvent(UserEvent event) 
        {
            var nameFromCache = await _cacheRepository.GetFullName(event.Id);
            // var nameFromDb = _dbRepository.GetFullName(event.Id);

            //var name = await Task.WhenAny(nameFromCache, nameFromDb);
            if (name is null) {
                _idsForBatching.Add(event);
            }
            event.SetFullName(name);

            await _eventSaver.Save(event);
        }
      

    }
    public interface IDbRepository
    {
        Task<Dictionary<long,string>> GetFullNames(long[] userIds); // Max batch 100
    }
    public interface ICacheRepository
    {
        Task<string?> GetFullName(long userId);
    }
    public interface IEventSaver
    {
        Task Save(UserEvent[] evs); // Max batch 50
    }
    public sealed class UserEvent
    {
        public readonly DateTime CreationDate;
        public readonly string Data;
        public readonly long Id;
        public string FullName { get; private set; }
        public void SetFullName(string fullName)
        {
            FullName = fullName;
        }
    }
}
