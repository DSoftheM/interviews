using System.Threading.Channels;  
  
await new Client().GetArrayAsync3(Enumerable.Range(1, 10).ToArray(), 3);  
  
public class Response  
{  
}  
  
public interface IClient  
{  
    Task<Response> GetAsync(int id);  
}  
  
public class Client : IClient  
{  
    public async Task<Response> GetAsync(int id)  
    {  
        await Task.Delay(1000);  
        Console.WriteLine($"{id} is working...");  
        return new Response();  
    }  
}  
  
public static class ClientExtensions  
{  
    public static async Task<Response[]> GetArrayAsync3(  
        this IClient client,  
        int[] ids,  
        int maxDegreeOfParallelism)  
    {  
        var results = new Response[ids.Length];  
        var channel = Channel.CreateBounded<(int index, int id)>(maxDegreeOfParallelism * 2);  
  
        var producerTask = StartProducer();  
  
        var workers = Enumerable.Range(0, maxDegreeOfParallelism)  
            .Select(_ => Worker())  
            .ToArray();  
  
        await Task.WhenAll(workers.Append(producerTask));  
        return results;  
  
        async Task StartProducer()  
        {  
            for (int i = 0; i < ids.Length; i++)  
                await channel.Writer.WriteAsync((i, ids[i]));  
  
            channel.Writer.Complete();  
        }  
  
        async Task Worker()  
        {  
            await foreach (var (index, id) in channel.Reader.ReadAllAsync())  
            {  
                results[index] = await client.GetAsync(id);  
            }  
        }  
    }  
  
    public static async Task<Response[]> GetArrayAsync2(  
        this IClient client,  
        int[] ids,  
        int maxDegreeOfParallelism)  
    {  
        var results = new Response[ids.Length];  
  
        foreach (var batch in ids  
                     .Select((id, i) => (id, i))  
                     .Chunk(maxDegreeOfParallelism))  
        {  
            var tasks = batch  
                .Select(x => client.GetAsync(x.id))  
                .ToArray();  
  
            var responses = await Task.WhenAll(tasks);  
  
            for (int i = 0; i < batch.Length; i++)  
                results[batch[i].i] = responses[i];  
        }  
  
        return results;  
    }  
  
    public static async Task<Response[]> GetArrayAsync(this IClient client, int[] ids, int maxDegreeOfParallelism)  
    {  
        var semaphore = new SemaphoreSlim(maxDegreeOfParallelism);  
        var tasks = new Task<Response>[ids.Length];  
  
        for (int i = 0; i < ids.Length; i++)  
        {  
            var id = ids[i];  
  
            tasks[i] = ExecuteAsync(client, semaphore, id);  
        }  
  
        return await Task.WhenAll(tasks);  
    }  
  
    private static async Task<Response> ExecuteAsync(  
        IClient client,  
        SemaphoreSlim semaphore,  
        int id)  
    {  
        await semaphore.WaitAsync();  
        try  
        {  
            return await client.GetAsync(id);  
        }  
        finally  
        {  
            semaphore.Release();  
        }  
    }  
}
