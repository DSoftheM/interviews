Реализовать метод который заполняет полное имя пользователя из хранилища(IDbRepository или ICacheRepository) и прокидывает обогащенное событие дальше(IEventSaver)
Необходимо добиться максимальной производительности
В кэше содержиться 99.9% актуальных данных
изначальный порядок событий при сохранении можно нарушать

```cs
namespace ConsoleApp
{
    public sealed class UserEnricher
    {
        private readonly IDbRepository _dbRepository;
        private readonly ICacheRepository _cacheRepository;
        private readonly IEventSaver _eventSaver;

        public UserEnricher(IDbRepository dbRepository, ICacheRepository cacheRepository, IEventSaver eventSaver)
        {
            _cacheRepository = cacheRepository;
            _eventSaver = eventSaver;
            _dbRepository = dbRepository;
        }

        public async Task Enrich(IEnumerable<UserEvent> events)
        {
        }
    }

    public interface IDbRepository
    {
        Task<string> GetFullName(long userId);
    }

    public interface ICacheRepository
    {
        Task<string> GetFullName(long userId);
    }

    public interface IEventSaver
    {
        Task Save(UserEvent ev);
    }

    public sealed class UserEvent
    {
        public readonly DateTime CreationDate;
        public readonly string Data;
        public readonly long Id;
        public string FullName { get; private set; }
        
        public void SetFullName(string fullName)
        {
            FullName = fullName;
        }
    }
}
```
